// Merge into existing server.js; don't create a second Express app.
require('dotenv').config();
const express = require('express');
const cors = require('cors');
const helmet = require('helmet');
const mongoSanitize = require('express-mongo-sanitize');
const { connectDB } = require('./config/db');
const { validateEnv } = require('./config/env');
const { apiLimiter } = require('./middleware/security');
const authRoutes = require('./routes/authRoutes');
const { notFound, errorHandler } = require('./middleware/errorHandler');

const app = express();

// Add after app = express(), before API routes:
app.disable('x-powered-by');
app.use(helmet());
app.use(cors({
  origin: process.env.CLIENT_ORIGIN
    ? process.env.CLIENT_ORIGIN.split(',').map(s => s.trim())
    : false,
  methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE'],
  allowedHeaders: ['Content-Type', 'Authorization'],
}));
app.use(express.json({ limit: '20kb' }));
app.use(mongoSanitize());
app.use(apiLimiter);
app.use('/api/auth', authRoutes);

// Optional example mounts only if matching models exist; merge/adapt instead of duplicating routes:
// app.use('/api/bookings', require('./routes/bookingRoutes.example'));
// app.use('/api/events', require('./routes/eventRoutes.example'));

// At bottom, after all routes:
app.use(notFound);
app.use(errorHandler);

// Replace the old app.listen with this startup:
validateEnv();
connectDB().then(() => {
  const PORT = process.env.PORT || 5000;
  app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
}).catch(err => {
  console.error('Startup failed:', err.message);
  process.exit(1);
});
